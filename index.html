const canvas = document.getElementById('starCanvas'); const ctx = canvas.getContext('2d'); let stars = []; function setup() { canvas.width = window.innerWidth; canvas.height = window.innerHeight; // Criamos as estrelas como objetos para podermos animar o brilho stars = []; const starCount = Math.floor((canvas.width * canvas.height) / 3000); for (let i = 0; i < starCount; i++) { stars.push({ x: Math.random() * canvas.width, y: Math.random() * canvas.height, size: Math.random() * 2 + 0.5, blink: Math.random() * 0.05, // Velocidade da pulsação opacity: Math.random() }); } } function draw() { ctx.clearRect(0, 0, canvas.width, canvas.height); stars.forEach(star => { // Faz o brilho "oscilar" para parecer neon real star.opacity += star.blink; if (star.opacity > 1 || star.opacity < 0.2) star.blink *= -1; // Camada 1: O brilho externo (Aura Neon) ctx.shadowBlur = 15 * star.opacity; ctx.shadowColor = "#00ffff"; // Camada 2: O brilho médio ctx.fillStyle = `rgba(0, 255, 255, ${star.opacity})`; ctx.beginPath(); ctx.arc(star.x, star.y, star.size, 0, Math.PI * 2); ctx.fill(); // Camada 3: O núcleo branco (o centro da lâmpada neon) ctx.shadowBlur = 0; ctx.fillStyle = "white"; ctx.beginPath(); ctx.arc(star.x, star.y, star.size * 0.5, 0, Math.PI * 2); ctx.fill(); }); requestAnimationFrame(draw); // Isso faz o neon brilhar e piscar em tempo real } window.addEventListener('resize', setup); setup(); draw();
